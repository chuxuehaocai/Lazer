cmake_minimum_required(VERSION 3.21)
project(lazer_alsa_route_probe LANGUAGES CXX)
enable_testing()
add_executable(lazer-alsa-route-probe probe.cpp)
target_include_directories(lazer-alsa-route-probe PRIVATE ../../src)
target_compile_features(lazer-alsa-route-probe PRIVATE cxx_std_20)
if(MSVC)
    target_compile_options(lazer-alsa-route-probe PRIVATE /W4 /WX /EHsc /utf-8)
else()
    target_compile_options(lazer-alsa-route-probe PRIVATE -Wall -Wextra -Werror)
endif()
add_test(NAME lazer-alsa-route-policy COMMAND lazer-alsa-route-probe)
set_tests_properties(lazer-alsa-route-policy PROPERTIES TIMEOUT 10)
